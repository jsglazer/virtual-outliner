% \DocTitle and \DocAuthor are set per export (the note's title: frontmatter or
% its file name, and the Author setting) in a small include that Virtual
% Outliner loads BEFORE this preamble, so these \providecommand lines only
% apply when this preamble is used on its own. An empty \DocTitle falls back
% to \jobname.
\providecommand{\DocTitle}{}
\providecommand{\DocAuthor}{}
\newcommand{\HeaderTitle}{\ifdefempty{\DocTitle}{\jobname}{\DocTitle}}
\fancyhead{} % clear all header fields
\fancyhead[L]{\small{\textbf{\HeaderTitle}}}
\fancyhead[R]{\small{\DocAuthor}}
\fancyfoot{}
\fancyfoot[L]{\scriptsize{\thepage\ of \pageref{LastPage}}}
\fancyfoot[R]{\scriptsize{\dashdate{\today} at \dottime}}
\renewcommand{\headrulewidth}{0.4pt}
\renewcommand{\footrulewidth}{0.4pt}
